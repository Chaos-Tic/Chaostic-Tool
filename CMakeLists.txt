cmake_minimum_required(VERSION 3.20)
project(ChaosticToolDesktop VERSION 1.2.0 LANGUAGES NONE)
set(CMAKE_INSTALL_LIBDIR lib CACHE PATH "Library directory (unused by this Python application)")
include(GNUInstallDirs)
set(CHAOSTIC_BUNDLE_DIR "" CACHE PATH "Optional frozen PyInstaller bundle for portable DEB/RPM packages")
if(CHAOSTIC_BUNDLE_DIR)
    if(NOT EXISTS "${CHAOSTIC_BUNDLE_DIR}/ChaosticTool")
        message(FATAL_ERROR "CHAOSTIC_BUNDLE_DIR must contain the tested ChaosticTool executable")
    endif()
    set(CHAOSTIC_EXEC "\"${CMAKE_INSTALL_FULL_LIBDIR}/chaostictool-desktop/ChaosticTool\"")
    install(DIRECTORY "${CHAOSTIC_BUNDLE_DIR}/" DESTINATION "${CMAKE_INSTALL_LIBDIR}/chaostictool-desktop" USE_SOURCE_PERMISSIONS)
else()
    set(CHAOSTIC_EXEC "/usr/bin/python3 -B \"${CMAKE_INSTALL_FULL_DATADIR}/chaostictool-desktop/chaostic_desktop.py\"")
    install(FILES chaostic_desktop.py DESTINATION "${CMAKE_INSTALL_DATADIR}/chaostictool-desktop")
    install(DIRECTORY desktop core modules DESTINATION "${CMAKE_INSTALL_DATADIR}/chaostictool-desktop"
            PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
endif()
configure_file(packaging/linux/launcher.in chaostictool-desktop @ONLY)
configure_file(packaging/linux/chaostictool-desktop.desktop.in chaostictool-desktop.desktop @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/chaostictool-desktop" DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/chaostictool-desktop.desktop" DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
install(FILES desktop/assets/icon.svg DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps" RENAME chaostictool-desktop.svg)
install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/chaostictool-desktop")
set(CPACK_PACKAGE_NAME chaostictool-desktop)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_CONTACT "Chaos-Tic <58691435+Chaos-Tic@users.noreply.github.com>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Desktop workspace for authorized security testing")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/Chaos-Tic/Chaostic-Tool")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
set(CPACK_RPM_PACKAGE_LICENSE MIT)
set(CPACK_RPM_PACKAGE_AUTOREQPROV OFF)
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /usr/share /usr/share/licenses /usr/share/applications /usr/share/icons /usr/share/icons/hicolor /usr/share/icons/hicolor/scalable /usr/share/icons/hicolor/scalable/apps)
if(CHAOSTIC_BUNDLE_DIR)
    execute_process(COMMAND getconf GNU_LIBC_VERSION OUTPUT_VARIABLE CHAOSTIC_GLIBC OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    string(REGEX REPLACE "^glibc " "" CHAOSTIC_GLIBC "${CHAOSTIC_GLIBC}")
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6 (>= ${CHAOSTIC_GLIBC}), libegl1, libopengl0, libxkbcommon0, libxkbcommon-x11-0, libxcb-cursor0, libxcb-icccm4, libxcb-keysyms1, libxcb-shape0, libxcb-xinerama0, libxcb-image0, libxcb-render-util0, libxcb-xkb1, libdbus-1-3, libfontconfig1, libglib2.0-0, libstdc++6, libgcc-s1")
    set(CPACK_RPM_PACKAGE_REQUIRES "glibc >= ${CHAOSTIC_GLIBC}, mesa-libEGL, libglvnd-opengl, libxkbcommon, libxkbcommon-x11, xcb-util-cursor, xcb-util-wm, xcb-util-keysyms, xcb-util-image, xcb-util-renderutil, libxcb, dbus-libs, fontconfig, glib2, libstdc++, libgcc")
else()
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE all)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3 (>= 3.11), python3-pyside6.qtwidgets (>= 6.8), python3-pyside6.qtsvg, python3-psutil, python3-dnspython, python3-py7zr, python3-pyte")
    set(CPACK_RPM_PACKAGE_ARCHITECTURE noarch)
    set(CPACK_RPM_PACKAGE_REQUIRES "python3 >= 3.11, python3-pyside6 >= 6.8, python3-psutil, python3-dns, python3-py7zr, python3-pyte")
endif()
include(CPack)
