cmake_minimum_required(VERSION 3.20)
project(ChaosticTool VERSION 1.5.0 LANGUAGES NONE)
# Pure Python application: CMake only stages files; dependencies belong to the package manager.
set(CMAKE_INSTALL_LIBDIR lib CACHE PATH "Library directory (unused by this Python application)")
include(GNUInstallDirs)
configure_file(packaging/launcher.in chaostictool @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/chaostictool" DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES chaostictool.py DESTINATION "${CMAKE_INSTALL_DATADIR}/chaostictool")
install(DIRECTORY core modules assets DESTINATION "${CMAKE_INSTALL_DATADIR}/chaostictool"
        PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/chaostictool")
set(CPACK_PACKAGE_NAME chaostictool)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_CONTACT "Chaos-Tic <58691435+Chaos-Tic@users.noreply.github.com>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Terminal interface for authorized security testing")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/Chaos-Tic/Chaostic-Tool")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE all)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3 (>= 3.11), python3-rich (>= 13.7), python3-prompt-toolkit (>= 3.0)")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_RPM_PACKAGE_ARCHITECTURE noarch)
set(CPACK_RPM_PACKAGE_LICENSE MIT)
set(CPACK_RPM_PACKAGE_REQUIRES "python3 >= 3.11, python3-rich >= 13.7, python3-prompt-toolkit >= 3.0")
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /usr/share /usr/share/licenses)
set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
include(CPack)
